\section*{Selected Projects}
\project{Interactive 3D Fluid Simulator}{\href{https://dimitrichrysafis.github.io/media/post10/index.html}{Live Demo} \textcolor{rulegray}{|} \href{https://github.com/DimitriChrysafis/3d-Volumetric-Fluid-Simulator}{GitHub} \textcolor{rulegray}{|} WebGPU, WGSL, JavaScript}{
\begin{itemize}
  \item Designed a browser-native MLS-MPM physics engine and GPU pipeline that simulates \textbf{400,000 particles} in real time using parallel particle-to-grid, grid-update, and grid-to-particle compute passes.
  \item Eliminated CPU-side particle copies with shared storage buffers and direct rendering; implemented 128-thread workgroups, atomic fixed-point accumulation, dynamic piston boundaries, turbulence pulses, and adaptive render density.

\project{Volumetric Sphere Packing for Arbitrary 3D Meshes}{\href{https://github.com/DimitriChrysafis/SpherePacker}{GitHub} \textcolor{rulegray}{|} \href{https://dimitrichrysafis.github.io/\#post/post8.md}{Technical Write-up} \textcolor{rulegray}{|} PyTorch, Open3D}{
\begin{itemize}
  \item Built a geometry pipeline that packs non-overlapping tangent spheres inside arbitrary OBJ meshes using M\"oller--Trumbore parity tests, nearest-surface queries, global grid seeding, and stochastic tangent growth.
  \item Vectorized containment, clearance, and overlap checks as batched PyTorch operations on CUDA/MPS; combined binary-search radius fitting, mesh decimation, surface sampling, and live Open3D visualization.
\end{itemize}}

\project{Tennis Match Prediction System}{CatBoost, Python, scikit-learn}{
\begin{itemize}
  \item Trained a gradient-boosted classifier on \textbf{92,000 ATP matches (1982--2024)}, engineering player and surface Elo, head-to-head history, recent form, and schedule-derived fatigue features.

\end{itemize}}
